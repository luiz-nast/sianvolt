# fica na net do LDR: o gêmeo acha ele pelo resistor (R9) e manda o "adc" com origem circuito. LED com 220 Ω
# no GPIO23; o LDR é o mesmo C10081 de pecas/ (o do placa_ldr.txt)
# (pinagem do DevKit de 30 pinos: 30 = GPIO23, 4 = GPIO34, 16 = 3V3, 14 e 17 = GND)

liga D1.2 GND        # − do LED no GND

liga R2.2 U1.4       # -> GPIO34 (ADC)
peca R9 resistor_axial 60 10 1k

liga R9.2 LDR1.1     # -> LDR
